def _process_single_merge_task(rule_row, connection_info):
    """
    Description:
        単一のマージルールに基づき、ファイルの読み込み・加工・マージ・保存を行う
        (main関数のループ内処理を切り出し)

    Args:
        rule_row (Series): マージルールの1行分のデータ
        connection_info (DataFrame): 接続先情報のマスタデータ

    Returns:
        なし

    Raises:
        KeyError: 必要な接続先情報が見つからない場合
        Exception: その他処理中のエラー

    Version:
        v1.0 (2026/01/21) [YourName] main関数からロジックを分離して作成
    """
    
    ### -----------------------------------------------------------
    ### ファイル情報の取得
    ### -----------------------------------------------------------
    # ルール行から各テーブル名に対応する接続情報を取得
    # 左テーブル、右テーブル、出力先テーブルの情報を抽出
    file_info1 = connection_info[connection_info[SYSTEM_NAME] == rule_row['左テーブル']].iloc[0]
    file_info2 = connection_info[connection_info[SYSTEM_NAME] == rule_row['右テーブル']].iloc[0]
    output_file_info = connection_info[connection_info[SYSTEM_NAME] == rule_row['出力先テーブル']].iloc[0]

    ### -----------------------------------------------------------
    ### データの読み込み
    ### -----------------------------------------------------------
    print_timestamp(f"{file_info1[SYSTEM_NAME]}のデータを読み込み中")
    df1 = _read_data(file_info1)
    
    print_timestamp(f"{file_info2[SYSTEM_NAME]}のデータを読み込み中")
    df2 = _read_data(file_info2)
    print_timestamp("データ読み込み完了")

    ### -----------------------------------------------------------
    ### カラムのフィルタリング (特定列のみ使用する場合)
    ### -----------------------------------------------------------
    # ※必要に応じてフィルタリングを行う。ここでは例としてハードコーディングされていたものを踏襲
    #   実運用では rule_row に「使用列」などの定義があればそれを使うのがベター
    df1 = _filter_columns(df1, '(UP技)\nUnit')
    df2 = _filter_columns(df2, '(PE技)\nUnit')

    ### -----------------------------------------------------------
    ### マージ条件の解析と適用
    ### -----------------------------------------------------------
    merge_keys1 = _extract_merge_keys(rule_row[MERGE_KEYS_SRC])
    merge_keys2 = _extract_merge_keys(rule_row[MERGE_KEYS_DEST])
    
    # 条件式がある場合はパースして適用
    merge_condition = None
    if CONDITIONS in rule_row and not pd.isna(rule_row[CONDITIONS]):
        merge_condition = _parse_merge_condition(rule_row[CONDITIONS])

    ### -----------------------------------------------------------
    ### データのマージ実行
    ### -----------------------------------------------------------
    print_timestamp("データをマージ中")
    merged_data = _merge_data(
        df1, df2, 
        merge_keys1, merge_keys2, 
        rule_row[MERGE_METHOD], 
        merge_condition
    )

    ### -----------------------------------------------------------
    ### 結果の出力（上書き保存）
    ### -----------------------------------------------------------
    _merge_data_to_existing_file(merged_data, output_file_info)
    print_timestamp("マージ処理完了")
